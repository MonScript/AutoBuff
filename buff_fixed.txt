local function saveConfig()
    ensureFolders()
    pcall(function()
        if writefile then
            local payload = {}
            for k, defVal in pairs(DefaultConfig) do
                if type(config[k]) == type(defVal) then
                    payload[k] = config[k]
                else
                    payload[k] = defVal
                end
            end
            writefile(CONFIG_FILE, HttpService:JSONEncode(payload))
        end
    end)
end

local function mergeConfigFromFile(path)
    if not (readfile and isfile) then
        return false
    end
    local okEx, exists = pcall(isfile, path)
    if not (okEx and exists) then
        return false
    end
    local okRd, raw = pcall(readfile, path)
    if not (okRd and type(raw) == "string" and raw ~= "") then
        return false
    end
    local okDec, data = pcall(function()
        return HttpService:JSONDecode(raw)
    end)
    if okDec and type(data) == "table" then
        for k, defVal in pairs(DefaultConfig) do
            if data[k] ~= nil and type(data[k]) == type(defVal) then
                config[k] = data[k]
            end
        end
        return true
    end
    return false
end

local function loadConfig()
    ensureFolders()

    if not loadedMain then
        local m1 = mergeConfigFromFile("MonSkillHub_Config.json")
        local m2 = mergeConfigFromFile("HideSpooferConfig.json")
        if m1 or m2 then
            saveConfig()
        end
    end
end

local function getUnitPosition(unit)
    if unit:IsA("BasePart") then
        return unit.Position
    elseif unit:IsA("Model") then
        local p = unit.PrimaryPart or unit:FindFirstChild("HumanoidRootPart") or unit:FindFirstChildWhichIsA("BasePart")
        if p then return p.Position end
    end
    return nil
end

local function getRukiaEnergy(unit)
    local unitPos = getUnitPosition(unit)
    local ignoreFolder = workspace:FindFirstChild("ignore")
    if ignoreFolder and unitPos then
        local closestDist = 8
        local foundVal = nil
        for _, obj in ipairs(ignoreFolder:GetChildren()) do
            if obj:IsA("BasePart") then
                local unitBar = obj:FindFirstChild("_unit_bar")
                if unitBar then
                    local dist = (Vector3.new(obj.Position.X, 0, obj.Position.Z) - Vector3.new(unitPos.X, 0, unitPos.Z)).Magnitude
                    if dist < closestDist then
                        local countLabel = unitBar:FindFirstChild("Count", true)
                        if countLabel and countLabel:IsA("TextLabel") then
                            local cur, maxVal = countLabel.Text:match("(%d+)%s*/%s*(%d+)")
                            if cur and maxVal then
                                local nCur = tonumber(cur)
                                local nMax = tonumber(maxVal)
                                if nCur and nMax and nMax > 0 then
                                    closestDist = dist
                                    foundVal = math.clamp(math.floor((nCur / nMax) * 100), 0, 100)
                                end
                            end
                        end
                    end
                end
            end
        end
        if foundVal then return foundVal end
    end
    return 0
end

local function getHakariEnergy(unit)

                        end
                    end
                end
            end
        end
        if foundVal then return foundVal end
    end

            local num = tonumber(desc.Text:match("(%d+)%%")) or tonumber(desc.Text:match("%d+"))

end

local function getMyUnitsByNames(namesList, maxCount)
    local units = {}
    local unitsFolder = workspace:FindFirstChild("_UNITS")
    if unitsFolder then
        for _, unit in ipairs(unitsFolder:GetChildren()) do
            local isMatch = false
            for _, name in ipairs(namesList) do
                if unit.Name:lower() == name:lower() then
                    isMatch = true
                    break
                end
            end
            if isMatch then
                local stats = unit:FindFirstChild("_stats")
                local isOwner = false
                if stats and stats:FindFirstChild("player") then
                    if stats.player.Value == LocalPlayer then isOwner = true end
                else
                    isOwner = true
                end
                if isOwner then table.insert(units, unit) end
            end
        end
    end

    table.sort(units, function(a, b)
        local oa = placementSeq[a] or 1e9
        local ob = placementSeq[b] or 1e9
        if oa ~= ob then return oa < ob end
        local posA = getUnitPosition(a) or Vector3.zero
        local posB = getUnitPosition(b) or Vector3.zero
        if posA.Z ~= posB.Z then return posA.Z < posB.Z end
        return posA.X < posB.X
    end)

    if maxCount and #units > maxCount then
        local trimmed = {}
        for i = 1, maxCount do trimmed[i] = units[i] end
        return trimmed
    end
    return units
end

local function EscapeRegex(str)
    return (str:gsub("([%^%$%(%)%%%.%[%]%*%+%-%?])", "%%%1"))
end

local function CaseInsensitive(str)
    local esc = EscapeRegex(str)
    return (esc:gsub("%a", function(c)
        return string.format("[%s%s]", c:lower(), c:upper())
    end))
end

local function FormatDmgText(val)
    if type(val) ~= "string" then
        return ""
    end
    local trimmed = val:match("^%s*(.-)%s*$")
    if trimmed == "" then
        return ""
    end
    if trimmed:lower():find("dmg", 1, true) then
        return trimmed
    end
    return trimmed .. " DMG"
end

local function GetFolderMediaEntries()
    ensureFolders()
    local entries = {}
    if not listfiles then
        return entries
    end
    local ok, rawList = pcall(listfiles, MEDIA_FOLDER)
    if not ok or type(rawList) ~= "table" then
        return entries
    end
    table.sort(rawList)
    for _, rawPath in ipairs(rawList) do
        local norm = rawPath:gsub("\\", "/")
        local baseName = norm:match("([^/]+)$") or norm
        local isDir = false
        if isfolder then
            local okD, dRes = pcall(isfolder, rawPath)
            if okD and dRes then
                isDir = true
            end
        end
        if isDir then
            table.insert(entries, {
                Path = rawPath,
                Name = "[GIF Folder] " .. baseName,
                Kind = "folder"
            })
        else
            local low = baseName:lower()
            if low:match("%.png$") or low:match("%.jpg$") or low:match("%.jpeg$") or low:match("%.webp$") or low:match("%.bmp$") or low:match("%.gif$") then
                table.insert(entries, {
                    Path = rawPath,
                    Name = "[Image] " .. baseName,
                    Kind = "image"
                })
            elseif low:match("%.webm$") or low:match("%.mp4$") then
                table.insert(entries, {
                    Path = rawPath,
                    Name = "[Video] " .. baseName,
                    Kind = "video"
                })
            end
        end
    end
    return entries
end

local function LoadCustomMediaFromPath(targetPath)
    CustomMediaState.Mode = "none"
    CustomMediaState.SingleAsset = ""
    CustomMediaState.Frames = {}
    CustomMediaState.CurrentFrame = 1
    CustomMediaState.Accumulator = 0
    CustomMediaState.Status = ""

    if type(targetPath) ~= "string" or targetPath == "" then
        CustomMediaState.Status = "No media selected (original guild icon)."
        return
    end

    local toAsset = ResolveCustomAssetFunc()
    if not toAsset then
        CustomMediaState.Status = "No asset loader available (need getcustomasset or getsynasset)."
        return
    end

    pcall(function()
        if isfolder and listfiles then
            local okFolder, existsFolder = pcall(isfolder, targetPath)
            if okFolder and existsFolder then
                local okList, files = pcall(listfiles, targetPath)
                if okList and type(files) == "table" then
                    table.sort(files)
                    local loadedFrames = {}
                    for _, filePath in ipairs(files) do
                        local low = filePath:lower()
                        if low:match("%.png$") or low:match("%.jpg$") or low:match("%.jpeg$") or low:match("%.webp$") or low:match("%.bmp$") then
                            local okA, assetUri = pcall(toAsset, filePath)
                            if okA and assetUri then
                                table.insert(loadedFrames, assetUri)
                            end
                        end
                    end
                    if #loadedFrames > 0 then
                        CustomMediaState.Mode = "frames"
                        CustomMediaState.Frames = loadedFrames
                        CustomMediaState.Status = "Loaded " .. #loadedFrames .. " frames (animated at GIF FPS)."
                        return
                    else
                        CustomMediaState.Status = "Folder has no usable frames (.png/.jpg/.webp/.bmp)."
                        return
                    end
                end
            end
        end

        if isfile then
            local okFile, existsFile = pcall(isfile, targetPath)
            if okFile and existsFile then
                local low = targetPath:lower()
                -- Náº¿u lÃ  file .gif ÄÆ¡n hoáº·c video .webm/.mp4
                if low:match("%.webm$") or low:match("%.mp4$") then
                    CustomMediaState.Status = "Local video can't play in Roblox. VideoFrame only accepts an uploaded rbxassetid:// video. Convert the GIF/video into a folder of PNG frames and pick the [GIF Folder] entry instead."
                    return
                else
                    -- Äá»i vá»i file .gif tÄ©nh hoáº·c áº£nh thÃ´ng thÆ°á»ng (.png, .jpg, .gif)
                    local okAsset, assetUri = pcall(toAsset, targetPath)
                    if okAsset and assetUri then
                        CustomMediaState.Mode = "image"
                        CustomMediaState.SingleAsset = assetUri
                        if low:match("%.gif$") then
                            CustomMediaState.Status = "Loaded .gif as a static image (Roblox does not animate .gif). For animation, put PNG frames in a folder and pick [GIF Folder]."
                        else
                            CustomMediaState.Status = "Loaded image."
                        end
                        return
                    else
                        CustomMediaState.Status = "Could not load this file. This executor may not support this format. Use .png/.jpg images or a folder of frames."
                        return
                    end
                end
            end
        end
    end)
end

local function HookTextLabel(obj, customResolver)
    if not obj or not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then
        return
    end
    if obj:IsDescendantOf(screenGui) then
        return
    end
    SaveOriginalText(obj)
    if customResolver then
        ResolversText[obj] = customResolver
    end
    local resolver = ResolversText[obj]
    if resolver then
        SetSafeText(obj, resolver(OriginalText[obj], obj))
    end

    if not HookedText[obj] then
        HookedText[obj] = true
        local conn = obj:GetPropertyChangedSignal("Text"):Connect(function()
            if Updating[obj] then
                return
            end
            OriginalText[obj] = obj.Text
            local activeResolver = ResolversText[obj]
            if activeResolver then
                SetSafeText(obj, activeResolver(OriginalText[obj], obj))
            end
        end)
        table.insert(SpoofConnections, conn)
    end
end

local function ApplyMediaOverlayToIcon(iconObj)
    if not iconObj or not iconObj.Parent or not iconObj:IsA("ImageLabel") then
        return
    end
    SaveOriginalImage(iconObj)

    if not config.SpoofGuild or CustomMediaState.Mode == "none" then
        ClearMediaOverlay(iconObj)
        if OriginalImage[iconObj] then
            SetSafeImage(iconObj, OriginalImage[iconObj])
        end
        return
    end

    local existing = MediaOverlays[iconObj]
    if existing and existing.Parent then
        if CustomMediaState.Mode == "video" and not existing:IsA("VideoFrame") then
            ClearMediaOverlay(iconObj)
            existing = nil
        elseif CustomMediaState.Mode ~= "video" and not existing:IsA("ImageLabel") then
            ClearMediaOverlay(iconObj)
            existing = nil
        end
    else
        existing = nil
    end

    if CustomMediaState.Mode == "video" then
        SetSafeImage(iconObj, "")
        local vid = existing or Instance.new("VideoFrame")
        vid.Name = "CustomGuildMediaOverlay"
        vid.Size = UDim2.fromScale(1, 1)
        vid.Position = UDim2.fromScale(0, 0)
        vid.BackgroundTransparency = 1
        vid.BorderSizePixel = 0
        vid.ZIndex = iconObj.ZIndex + 5
        vid.Video = CustomMediaState.SingleAsset
        vid.Looped = true
        vid.Volume = 0
        vid.Playing = true
        vid.Parent = iconObj
        pcall(function()
            vid:Play()
        end)
        MediaOverlays[iconObj] = vid
    elseif CustomMediaState.Mode == "image" then
        SetSafeImage(iconObj, CustomMediaState.SingleAsset)

local function ApplyInGameAndResultsSpoof()
    local pGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not pGui then

    end

    -- Trong ván: chỉ hiển thị giá trị, không thêm hậu tố DMG.
    -- Màn hình kết quả: giữ định dạng DMG như game gốc.
    local function FormatScoreForDisplay(val, showDmg)
        if type(val) ~= "string" then
            return ""
        end

        local trimmed = val:match("^%s*(.-)%s*$")
        if trimmed == "" then
            return ""
        end

        if showDmg then
            return FormatDmgText(trimmed)
        end

        return (trimmed:gsub("%s*[Dd][Mm][Gg]%s*$", ""))
    end

    local function ResolveHighScore(showDmg)
        return function(orig)
            if not config.SpoofInGameScores then
                return orig
            end
            local rawVal = (config.FakeHighScore ~= "" and config.FakeHighScore) or config.FakeScore
            local formatted = FormatScoreForDisplay(rawVal, showDmg)
            return formatted ~= "" and formatted or orig
        end
    end

    local function ResolveCurrentScore(showDmg)
        return function(orig)
            if not config.SpoofInGameScores or config.FakeCurrentScore == "" then
                return orig
            end
            local formatted = FormatScoreForDisplay(config.FakeCurrentScore, showDmg)
            return formatted ~= "" and formatted or orig
        end
    end

    local tourneyScoreUI = pGui:FindFirstChild("TournamentScoreUI")
    if tourneyScoreUI then
        local main = tourneyScoreUI:FindFirstChild("Main")
        if main then
            local highV = main:FindFirstChild("HighScore") and main.HighScore:FindFirstChild("V")
            if highV then
                HookTextLabel(highV, ResolveHighScore(false))
            end
            local scoreV = main:FindFirstChild("Score") and main.Score:FindFirstChild("V")
            if scoreV then
                HookTextLabel(scoreV, ResolveCurrentScore(false))
            end
        end
    end

    local resultsUI = pGui:FindFirstChild("ResultsUI")
    if resultsUI then
        local middle = resultsUI:FindFirstChild("Holder") and resultsUI.Holder:FindFirstChild("Middle")
        if middle then
            local highV = middle:FindFirstChild("HighScore") and middle.HighScore:FindFirstChild("V")
            if highV then
                HookTextLabel(highV, ResolveHighScore(true))
            end
            local scoreV = middle:FindFirstChild("Score") and middle.Score:FindFirstChild("V")
            if scoreV then
                HookTextLabel(scoreV, ResolveCurrentScore(true))
            end
        end
    end

    local tourneyRanking = pGui:FindFirstChild("TournamentRankingUI")
    if tourneyRanking then
        local hsV = tourneyRanking:FindFirstChild("LevelSelect")
            and tourneyRanking.LevelSelect:FindFirstChild("InfoFrame")
            and tourneyRanking.LevelSelect.InfoFrame:FindFirstChild("ScoreInfo")
            and tourneyRanking.LevelSelect.InfoFrame.ScoreInfo:FindFirstChild("HighScore")
            and tourneyRanking.LevelSelect.InfoFrame.ScoreInfo.HighScore:FindFirstChild("V")
        if hsV then
            HookTextLabel(hsV, ResolveHighScore(true))
        end
    end

    local tourneyResults = pGui:FindFirstChild("TournamentResultsUI")
    if tourneyResults then
        local hsV = tourneyResults:FindFirstChild("Main")
            and tourneyResults.Main:FindFirstChild("Wrapper")
            and tourneyResults.Main.Wrapper:FindFirstChild("TournamentHolder")
            and tourneyResults.Main.Wrapper.TournamentHolder:FindFirstChild("ScoreInfo")
            and tourneyResults.Main.Wrapper.TournamentHolder.ScoreInfo:FindFirstChild("HighScore")

local function ProcessSingleLeaderboard(board)
    if not board then
        return
    end
    local container = board:FindFirstChild("Main")
        and board.Main:FindFirstChild("Wrapper")
        and board.Main.Wrapper:FindFirstChild("Container")
    if not container then
        return
    end

    local topScroll = container:FindFirstChild("Top")
    local yourRow = container:FindFirstChild("YourRow")

    if not config.SpoofTournament then
        if topScroll then
            for _, child in ipairs(topScroll:GetChildren()) do
                if child:IsA("Frame") then
                    RestoreBoardRowOriginal(child)
                end
            end
        end
        if yourRow then
            RestoreBoardRowOriginal(yourRow)
        end
        return
    end

    local orderedFrames = {}
    if topScroll then
        local spotlight = topScroll:FindFirstChild("Top")
        if spotlight and spotlight:IsA("Frame") then
            table.insert(orderedFrames, {Frame = spotlight, IsSpotlight = true, SortRank = 1})
        end
        local listRows = {}
        for _, child in ipairs(topScroll:GetChildren()) do
            if child:IsA("Frame") and child.Name == "LeaderboardFrameLevelSelect" then
                local pLbl = child:FindFirstChild("Place")
                if pLbl then
                    SaveOriginalText(pLbl)
                end
                local origPlace = pLbl and OriginalText[pLbl] or "#999"
                local rankNum = tonumber(origPlace:match("%d+")) or 999
                table.insert(listRows, {Frame = child, IsSpotlight = false, SortRank = rankNum})
            end
        end
        table.sort(listRows, function(a, b)
            if a.SortRank ~= b.SortRank then
                return a.SortRank < b.SortRank
            end
            return a.Frame.LayoutOrder < b.Frame.LayoutOrder
        end)
        for _, entry in ipairs(listRows) do
            table.insert(orderedFrames, entry)
        end
    end

    local realEntries = {}
    for _, item in ipairs(orderedFrames) do
        local f = item.Frame
        local uLbl = f:FindFirstChild("Username")
        local aLbl = f:FindFirstChild("Amount")
        local pLbl = f:FindFirstChild("Place")
        local iLbl = f:FindFirstChild("PlayerIcon")
        if uLbl then SaveOriginalText(uLbl) end
        if aLbl then SaveOriginalText(aLbl) end
        if pLbl then SaveOriginalText(pLbl) end
        if iLbl then SaveOriginalImage(iLbl) end

        local origUser = uLbl and OriginalText[uLbl] or ""
        local isSelf = origUser:lower():find(RealUsername:lower(), 1, true) ~= nil
        if not isSelf then
            table.insert(realEntries, {
                IsLocal = false,
                Username = "@Unknown",
                Amount = "No. No. No",
                Icon = "rbxthumb://type=AvatarHeadShot&id=1&w=420&h=420"
            })
        end
    end

    local targetRank = tonumber(config.FakePlace:match("%d+")) or 1
    if targetRank < 1 then
        targetRank = 1
    end

    local localEntry = {
        IsLocal = true,
        Username = "@Unknown",
        Amount = config.FakeScore ~= "" and config.FakeScore or "0",
        Icon = "rbxthumb://type=AvatarHeadShot&id=" .. (config.FakeUserId ~= "" and config.FakeUserId or "1") .. "&w=420&h=420"
    }

    local insertIdx = math.min(targetRank, #realEntries + 1)
    table.insert(realEntries, insertIdx, localEntry)

    for idx, item in ipairs(orderedFrames) do
        local slotData = realEntries[idx]
        if slotData then
            local rowPayload = {
                IsLocal = slotData.IsLocal,
                Place = "#" .. tostring(idx),
                Username = slotData.Username,
                Amount = slotData.Amount,
                Icon = slotData.Icon
            }
            ApplyBoardRowData(item.Frame, rowPayload, item.IsSpotlight)
        end
    end

    if yourRow and yourRow:IsA("Frame") then
        local yourPayload = {
            IsLocal = true,
            Place = "#" .. tostring(targetRank),
            Username = "@Unknown",
            Amount = config.FakeScore ~= "" and config.FakeScore or "0",
            Icon = "rbxthumb://type=AvatarHeadShot&id=" .. (config.FakeUserId ~= "" and config.FakeUserId or "1") .. "&w=420&h=420"
        }
        ApplyBoardRowData(yourRow, yourPayload, false)
    end
end

UIBg.applyBackground = function()
    local showBg = (config.Mica == true) and (UIBg.state.Mode ~= "none")
    UIBg.img.Visible = showBg
    if not showBg then
        return
    end
    local img
    if UIBg.state.Mode == "frames" and #UIBg.state.Frames > 0 then
        img = UIBg.state.Frames[UIBg.state.CurrentFrame] or UIBg.state.Frames[1]
    elseif UIBg.state.Mode == "image" then
        img = UIBg.state.SingleAsset
    end
    if img then
        UIBg.img.Image = img
    end
    local opacity = tonumber(config.UIBgOpacity)
    if type(opacity) ~= "number" then
        opacity = 75
    end
    if opacity < 0 then opacity = 0 end
    if opacity > 100 then opacity = 100 end
    UIBg.img.ImageTransparency = 1 - (opacity / 100)
end

UIBg.originals = {}
UIBg.applyMica = function()
    local on = config.Mica == true
    for _, obj in ipairs(mainFrame:GetDescendants()) do
        if obj:IsA("Frame") or obj:IsA("ScrollingFrame") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            if on then
                if UIBg.originals[obj] == nil then
                    UIBg.originals[obj] = obj.BackgroundTransparency
                end
                local orig = UIBg.originals[obj]
                local t = orig + (1 - orig) * 0.5
                if obj.BackgroundTransparency ~= t then
                    obj.BackgroundTransparency = t
                end
            else
                if UIBg.originals[obj] ~= nil then
                    obj.BackgroundTransparency = UIBg.originals[obj]
                end
            end
        end
    end
    UIBg.applyBackground()
end

local function styleButton(btn, color)
    local accent = color:Lerp(Color3.new(1, 1, 1), 0.3)
    local baseSize = btn.Size
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = accent }):Play()
        TweenService:Create(btn, TweenInfo.new(0.12), { Size = UDim2.new(baseSize.X.Scale, baseSize.X.Offset + 3, baseSize.Y.Scale, baseSize.Y.Offset + 2) }):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = color }):Play()
        TweenService:Create(btn, TweenInfo.new(0.12), { Size = baseSize }):Play()
    end)
    btn.MouseButton1Click:Connect(playClick)
end

task.spawn(function()
    while true do
        local units = getMyUnitsByNames(RUKIA_NAMES, RUKIA_REQUIRED)
        updateRukiaDisplay(units)

        if isRukiaRunning then
            if #units < RUKIA_REQUIRED then
                rukiaUI.countdownLabel.Text = "Cooldown: Paused"
                rukiaUI.stateLabel.Text = string.format("Placed: %d/%d Units", #units, RUKIA_REQUIRED)
                currentActiveRukia = nil
                rukiaActiveUntil = 0
                task.wait(0.5)
            else
                local now = os.clock()
                if currentActiveRukia and currentActiveRukia.Parent and currentActiveRukia:IsDescendantOf(workspace) and rukiaActiveUntil > now then
                    local remain = rukiaActiveUntil - now
                    rukiaUI.countdownLabel.Text = string.format("Freeze: %.1fs left", remain)
                    rukiaUI.stateLabel.Text = "Active Freeze Buffing"
                    task.wait(0.1)
                else
                    currentActiveRukia = nil
                    local unitData = {}
                    for _, u in ipairs(units) do
                        table.insert(unitData, {
                            unit = u,
                            energy = getRukiaEnergy(u)
                        })
                    end

                    table.sort(unitData, function(a, b)
                        return a.energy > b.energy
                    end)

                    local bestUnit = unitData[1].unit
                    local bestEnergy = unitData[1].energy
                    local secondEnergy = unitData[2] and unitData[2].energy or 0

                    if bestEnergy < 100 then
                        rukiaUI.countdownLabel.Text = "Cooldown: Ready"
                        rukiaUI.stateLabel.Text = string.format("Charging: Max %d%%", bestEnergy)
                        task.wait(0.1)
                    elseif secondEnergy < 50 then
                        rukiaUI.countdownLabel.Text = "Cooldown: Ready"
                        rukiaUI.stateLabel.Text = string.format("Sync: 2nd at %d%% (Need 50%%)", secondEnergy)
                        task.wait(0.1)
                    else
                        local castSucceeded = castSkill(bestUnit)

                        if castSucceeded then
                            currentActiveRukia = bestUnit
                            rukiaActiveUntil = os.clock() + RUKIA_DURATION
                            rukiaUI.countdownLabel.Text = string.format("Freeze: %.1fs left", RUKIA_DURATION)
                            rukiaUI.stateLabel.Text = "Freeze command sent"
                            task.wait(0.3)
                        else
                            currentActiveRukia = nil
                            rukiaActiveUntil = 0
                            rukiaUI.countdownLabel.Text = "Cooldown: Ready"
                            rukiaUI.stateLabel.Text = "Skill request failed; retrying"
                            task.wait(0.5)
                        end
                    end
                end
            end
        else
            rukiaUI.countdownLabel.Text = "Cooldown: Stopped"
            rukiaUI.stateLabel.Text = "Status: OFF"
            task.wait(0.3)
        end
    end
end)

local function executeErwinStep(stepIndex)
    if not isErwinRunning then return end

        erwinUI.stateLabel.Text = "Unit #1 buffs units 2, 3, and 4"
    elseif stepIndex == 2 then
        erwinUI.stateLabel.Text = "Unit #2 buffs units 1, 3, and 4"
    elseif stepIndex == 3 then

    end

local function WatchScreenGuiEnable(sg)
    if not sg or not sg:IsA("ScreenGui") or WatchedGuis[sg] then
        return
    end
    WatchedGuis[sg] = true
    table.insert(SpoofConnections, sg:GetPropertyChangedSignal("Enabled"):Connect(function()
        if sg.Enabled then
            task.defer(function()
                ApplySpecificPlayerGuiSpoof()
            end)
        end
    end))
end

local function BindSpoofDynamicListeners()
    local wsTourney = workspace:FindFirstChild("_TOURNAMENT")
    if wsTourney then
        table.insert(SpoofConnections, wsTourney.DescendantAdded:Connect(function(desc)
            if desc:IsA("Frame") and (desc.Name == "LeaderboardFrameLevelSelect" or desc.Name == "Top" or desc.Name == "YourRow") then
                task.defer(function()
                    ApplyTournamentBoardsSpoof()
                end)
            end
        end))
    end

    local pGui = LocalPlayer:WaitForChild("PlayerGui")
    for _, child in ipairs(pGui:GetChildren()) do
        if child.Name == "ResultsUI" or child.Name == "TournamentScoreUI" or child.Name == "TournamentResultsUI" or child.Name == "TournamentRankingUI" or child.Name == "PartyUI" then
            WatchScreenGuiEnable(child)
        end
    end

    table.insert(SpoofConnections, pGui.ChildAdded:Connect(function(child)
        if child.Name == "ResultsUI" or child.Name == "TournamentScoreUI" or child.Name == "TournamentResultsUI" or child.Name == "TournamentRankingUI" or child.Name == "PartyUI" then
            WatchScreenGuiEnable(child)
            task.defer(function()
                ApplySpecificPlayerGuiSpoof()
            end)
        end
    end))

    table.insert(SpoofConnections, pGui.DescendantAdded:Connect(function(desc)
        if desc:IsDescendantOf(screenGui) then
            return
        end
        if desc:IsA("Frame") and (desc.Name == "LeaderboardFrameLevelSelect" or desc.Name == "Top" or desc.Name == "YourRow") then
            task.defer(function()
                ApplyTournamentBoardsSpoof()
            end)
        elseif desc.Name == "PlayerFrame" and desc.Parent and desc.Parent.Name == "Members" then
            task.defer(function()
                ApplySpecificPlayerGuiSpoof()
            end)
        elseif desc.Name == "TournamentScoreUI" or desc.Name == "ResultsUI" or desc.Name == "TournamentResultsUI" or desc.Name == "HighScore" or desc.Name == "Score" then
            task.defer(function()
                ApplyInGameAndResultsSpoof()
            end)
        end
    end))
end

            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Chat, false)

        CustomMediaState.Accumulator = CustomMediaState.Accumulator + dt
        if CustomMediaState.Accumulator >= interval then
            CustomMediaState.Accumulator = CustomMediaState.Accumulator % interval
            CustomMediaState.CurrentFrame = (CustomMediaState.CurrentFrame % #CustomMediaState.Frames) + 1
            local nextUri = CustomMediaState.Frames[CustomMediaState.CurrentFrame]
            for iconObj, overlay in pairs(MediaOverlays) do
                if iconObj and iconObj.Parent and overlay and overlay.Parent and overlay:IsA("ImageLabel") then
                    SetSafeImage(iconObj, nextUri)
                    overlay.Image = nextUri
                end
            end
        end
    end
    if config.Mica and UIBg.state.Mode == "frames" and #UIBg.state.Frames > 1 then
        local interval = 1 / math.max(1, config.GifFps)
        UIBg.state.Accumulator = UIBg.state.Accumulator + dt
        if UIBg.state.Accumulator >= interval then
            UIBg.state.Accumulator = UIBg.state.Accumulator % interval
            UIBg.state.CurrentFrame = (UIBg.state.CurrentFrame % #UIBg.state.Frames) + 1
            if UIBg.img and UIBg.img.Visible then
                UIBg.img.Image = UIBg.state.Frames[UIBg.state.CurrentFrame]
            end
        end
    end
end))
